# full launch once the dry run looks healthy
sbatch submit_anchor.sbatch

Monitor

squeue --me                                              # running/pending
sacct -j <jobid> -X --format=JobID,State,ExitCode,Elapsed
sacct -j <jobid> -X -n --format=State | sort | uniq -c   # tally by state
grep -iE "error|traceback|cuda|fail" logs/mamma_thuman_*_*.err
scancel <jobid>
